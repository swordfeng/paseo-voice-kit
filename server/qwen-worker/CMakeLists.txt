cmake_minimum_required(VERSION 3.18)
project(voicekit-qwen-worker LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
if(APPLE)
  set(CMAKE_INSTALL_RPATH "@loader_path")
else()
  set(CMAKE_INSTALL_RPATH "$ORIGIN")
endif()
set(CMAKE_BUILD_WITH_INSTALL_RPATH ON)
add_subdirectory("${QWEN_SOURCE_DIR}" qwen EXCLUDE_FROM_ALL)
target_include_directories(qwen-core PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/qwen")
add_executable(voicekit-qwen-worker worker.cpp)
target_include_directories(voicekit-qwen-worker PRIVATE "${QWEN_SOURCE_DIR}/src" "${QWEN_SOURCE_DIR}/vendor/yyjson" "${QWEN_SOURCE_DIR}/ggml/include")
target_link_libraries(voicekit-qwen-worker PRIVATE qwen-core yyjson)
